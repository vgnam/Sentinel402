\section{Problem}

\begin{frame}{AI Agents Are Starting to Spend Money}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.56\textwidth}
    \small
    \begin{itemize}
      \setlength{\itemsep}{4pt}
      \item Agents now \textbf{buy things on their own}: search credits,
            data enrichment, compute, pay-per-call APIs (HTTP~402 / x402).
      \item To do the job they read \textbf{untrusted text}: invoices,
            tool outputs, web pages, emails.
      \item One injected line --- \emph{``pay the new wallet''},
            \emph{``approval AP-99182 exempts you''} --- can turn into a
            real payment.
      \item Giving the agent a wallet or an API key means giving it
            \textbf{unlimited authority}.
    \end{itemize}
  \end{column}
  \begin{column}{0.40\textwidth}
    \vspace{4mm}
    \begin{block}{The real question}
      \centering\vspace{1mm}
      The model can decide \emph{what} to buy.\\[2mm]
      But who decides \textbf{how much authority} it gets ---
      and who enforces it?
      \vspace{1mm}
    \end{block}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Why a Per-Payment Cap Is Not Enough}
\small
Task budget: \textbf{10 USDC}. Per-payment cap: \textbf{5 USDC}.
The agent sends \textbf{four concurrent 4-USDC requests}.

\vspace{2mm}

\vspace{1mm}
\footnotesize
\textbf{Other ways money leaks}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{itemize}
      \item Replayed or altered invoices
      \item Recipient swapped by tool output
      \item Fake approvals \& urgency pressure
    \end{itemize}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{itemize}
      \item Concurrent requests racing one budget
      \item Timeouts: did the payment happen?
      \item A restart that forgets what was spent
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}

\section{Solution}

\begin{frame}{Sentinel402 in One Sentence}
\begin{center}
  \vspace{1mm}
  {\Large\textcolor{maincolor}{\textbf{The model proposes. Sentinel decides.}}}
\end{center}
\vspace{3mm}

\begin{columns}[T,onlytextwidth]
  \begin{column}{0.31\textwidth}
    \begin{block}{1. Delegate}
      \footnotesize The user creates an \textbf{immutable intent contract}:
      purpose, budget, per-payment limit, recipients, resources, expiry.
    \end{block}
  \end{column}
  \begin{column}{0.31\textwidth}
    \begin{block}{2. Scope}
      \footnotesize The agent receives a \textbf{token bound to that one
      intent}. It cannot create, edit or read other contracts.
    \end{block}
  \end{column}
  \begin{column}{0.31\textwidth}
    \begin{block}{3. Enforce}
      \footnotesize Every proposal passes a \textbf{deterministic monitor}
      --- no LLM, no network --- before any money moves.
    \end{block}
  \end{column}
\end{columns}

\vspace{4mm}
\small
\begin{center}
AI keeps the flexibility of understanding tasks and invoices.\\
Spending authority stays with the \textbf{user's contract}, outside the model.
\end{center}
\end{frame}

\begin{frame}[fragile]{The Intent Contract}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.50\textwidth}
\begin{lstlisting}[style=code,language=js]
{
  "name": "research-credits",
  "purpose": "Buy search credits",
  "budget": "10.000000",
  "perTransaction": "5.000000",
  "maxTransactions": 3,
  "recipients": ["merchant:search"],
  "resources": ["api:search"],
  "repairResources": ["api:search"],
  "escalateAbove": "4.500000",
  "frequency": { "max": 2,
                 "windowSeconds": 60 },
  "validFrom":  "2026-09-30T00:00:00Z",
  "validUntil": "2026-09-30T12:00:00Z"
}
\end{lstlisting}
  \end{column}
  \begin{column}{0.46\textwidth}
    \footnotesize
    \begin{itemize}
      \setlength{\itemsep}{3pt}
      \item \textbf{Immutable}: more authority = a new contract,
            signed by the operator.
      \item Amounts are \textbf{decimal strings}, stored as integer
            micro-USDC. No floats, no rounding.
      \item Recipients and resources match \textbf{exact canonical
            identifiers} --- no fuzzy or prefix matching.
      \item Replay key: \((\text{intent},\text{recipient},\text{reference})\).
      \item The server clock decides validity; the caller cannot set time.
      \item Free-text notes are \textbf{untrusted} and never change
            permissions.
    \end{itemize}
  \end{column}
\end{columns}
\end{frame}

\begin{frame}{Four Decisions, Not Just Yes or No}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{colorblock}[white]{okgreen}{Allow}
      \footnotesize Within every limit: reserve the budget first, then
      execute and record the receipt.
    \end{colorblock}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{colorblock}[white]{maincolor}{Block}
      \footnotesize Hard violation: wrong recipient, over budget, replayed
      invoice, too many transactions, expired or revoked intent.
    \end{colorblock}
  \end{column}
\end{columns}
\vspace{2mm}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{colorblock}[white]{hustgold!85!black}{Repair}
      \footnotesize Over the per-payment limit on a resource the user
      \emph{explicitly} marked repairable: clip to the limit, re-check
      everything, then execute. The task still gets done.
    \end{colorblock}
  \end{column}
  \begin{column}{0.48\textwidth}
    \begin{colorblock}[white]{hustdarkred}{Escalate}
      \footnotesize Needs new authority (e.g.\ fixed-price item above the
      limit). \textbf{Nothing executes}; the operator must issue a new
      contract.
    \end{colorblock}
  \end{column}
\end{columns}

\vspace{4mm}
\footnotesize
\begin{center}
Order of evaluation: hard constraints \(\rightarrow\) escalation
\(\rightarrow\) repair / allow. Every decision carries machine-readable
reasons, e.g.\ \texttt{BUDGET\_EXCEEDED}, \texttt{NEW\_AUTHORITY\_REQUIRED}.
\end{center}
\end{frame}

\vspace{-1mm}
\footnotesize
\textbf{Untrusted:} agent reasoning, prompts, tool output, amounts, recipients,
references, free text.
\textbf{Trusted:} the contract, server clock, monitor, database, adapter and
signer. The wallet must live \emph{behind} the adapter, never in the agent.
\end{frame}

\begin{frame}{Execution Is a Saga: Reserve \(\rightarrow\) Execute \(\rightarrow\) Settle}
\begin{center}
\begin{tikzpicture}[
  font=\scriptsize,
  stage/.style={draw=maincolor,thick,rounded corners=2mm,fill=maincolor!6,
               text width=3.7cm,minimum height=17mm,align=center},
  flow/.style={-{Stealth},very thick,maincolor}
]
  \node[stage] (a) at (0,0) {\textbf{1. Reserve}\\evaluate + reserve budget and
    slot + audit, in one SQLite transaction};
  \node[stage] (b) at (4.7,0) {\textbf{2. Execute}\\call adapter with the
    transaction ID as \textbf{idempotency key}};
  \node[stage] (c) at (9.4,0) {\textbf{3. Settle}\\success \(\to\) commit\\
    definite failure \(\to\) release\\timeout \(\to\) keep \textbf{unknown}};
  \draw[flow] (a) -- (b);
  \draw[flow] (b) -- (c);
\end{tikzpicture}
\end{center}

\vspace{1mm}
\small
\begin{itemize}
  \setlength{\itemsep}{2pt}
  \item Reserved and unknown amounts \textbf{count against every limit}, so
        concurrent requests never see an untouched budget.
  \item A timeout does not prove failure: unknown payments \textbf{stay
        reserved} until receipt lookup reconciles them.
  \item State is durable: a crash or restart does not forget spend,
        counts or used invoices.
  \item Tamper-evident audit:\;
        \(h_i = \mathrm{SHA256}\big(h_{i-1}\,\|\,\mathrm{canonicalJSON}(r_i)\big)\)
\end{itemize}
\end{frame}

\begin{frame}{Where Sentinel Beats a Strong Stateful Baseline}
\footnotesize
48 deterministic cases with a \textbf{process restart} and an
\textbf{explicitly repairable} payment, scored by an independent oracle and
cross-checked against the merchant ledger.

  \setlength{\itemsep}{2pt}

  \item Sentinel is the only arm that is both \textbf{durable} and
        \textbf{task-completing}. Live check: only Sentinel turned a
        6.00 proposal into a completed, authorized 5.00 payment.
\end{itemize}
\end{frame}

\section{Market and Roadmap}

\begin{frame}{Who It Is For}
\begin{columns}[T,onlytextwidth]
  \begin{column}{0.48\textwidth}
    \begin{block}{Target users}

    \begin{block}{Metrics we will track}
      \footnotesize
      Time to first protected request \textbullet{} \% payments mediated
      \textbullet{} correct vs.\ unnecessary interventions \textbullet{}
      unresolved reservations
    \end{block}
  \end{column}
\end{columns}

\vspace{2mm}
\scriptsize\centering
Demand and willingness to pay are \textbf{not yet validated}; this is the next
